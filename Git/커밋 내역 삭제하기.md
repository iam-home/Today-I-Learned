1. 파이썬 설치
2. pip install git-filter-repo
3. echo '삭제할문자열==>변경할문자열' > replace.txt
4. git filter-repo --replace-text replace.txt --force
5. rm -rf replace.txt
6. git remote -v
7. git remote add origin 리포지토리주소
8. git push origin main --force
